// Init Telegram WebApp const tg = window.Telegram.WebApp; tg.expand(); async function refreshData() { const btn = document.querySelector('.btn'); btn.innerText = '⏳ Updating...'; try { // API URL ya Backend yako kwenye Render const response = await fetch('https://v4-engine.onrender.com/api/data'); const data = await response.json(); // 1. Update Pair & Direction document.getElementById('pair').innerText = data.pair || '--'; document.getElementById('direction').innerText = data.direction || '--'; // 2. Update Badge Color const dirElement = document.getElementById('direction'); if (data.direction && data.direction.toUpperCase() === 'BUY') { dirElement.className = 'badge badge-buy'; } else if (data.direction && data.direction.toUpperCase() === 'SELL') { dirElement.className = 'badge badge-sell'; } else { dirElement.className = 'badge badge-warning'; } btn.innerText = '🔄 Refresh'; } catch (error) { console.error("Error fetching data:", error); btn.innerText = '❌ Error!'; } }
